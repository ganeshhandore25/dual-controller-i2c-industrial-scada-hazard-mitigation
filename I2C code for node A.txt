
#include <Wire.h>

#define MQ2 A0
#define TMP36 A1
#define LDR A2

void setup()
{
  Serial.begin(9600);

  Wire.begin(8);
  Wire.onRequest(sendData);
}

void loop()
{
}

void sendData()
{
  int gas = analogRead(MQ2);
  int temp = analogRead(TMP36);
  int light = analogRead(LDR);

  Wire.write(highByte(gas));
  Wire.write(lowByte(gas));

  Wire.write(highByte(temp));
  Wire.write(lowByte(temp));

  Wire.write(highByte(light));
  Wire.write(lowByte(light));
}
