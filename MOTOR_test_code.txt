#define ENA 5
#define IN1 6
#define IN2 7

void setup()
{
  pinMode(ENA, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
}

void loop()
{
  // Motor ON
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  analogWrite(ENA, 200);
  delay(3000);

  // Motor OFF
  analogWrite(ENA, 0);
  delay(3000);
}
