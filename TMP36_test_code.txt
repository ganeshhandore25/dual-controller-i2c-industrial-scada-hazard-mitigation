#define TMP36 A1

float voltage;
float temperature;

void setup()
{
  Serial.begin(9600);
}

void loop()
{
  int sensorValue = analogRead(TMP36);

  voltage = sensorValue * (5.0 / 1023.0);

  temperature = (voltage - 0.5) * 100.0;

  Serial.print("Temperature: ");
  Serial.print(temperature);
  Serial.println(" C");

  delay(1000);
}
